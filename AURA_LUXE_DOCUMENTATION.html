4. Core Code Functions Explained

Function: formatPrice(amountInUSD)

Dynamically converts and formats prices according to active currency (USD, EUR, GBP, INR) with international locale rules.

const formatPrice = (amountInUSD) => {
  const rateObj = CURRENCY_RATES[currency] || CURRENCY_RATES.USD;
  const converted = amountInUSD * rateObj.rate;
  if (currency === 'INR') {
    return `₹${Math.round(converted).toLocaleString('en-IN')}`;
  }
  return `${rateObj.symbol}${converted.toLocaleString('en-US', {
    minimumFractionDigits: 0,
    maximumFractionDigits: 0
  })}`;
};
Function: exportData(type, format)

Generates client-side CSV or JSON files from memory and triggers browser file downloads with zero backend dependency.

const exportData = (type = 'orders', format = 'csv') => {
  let dataset = mapDatasetByType(type);
  if (format === 'json') {
    const jsonStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(dataset, null, 2));
    triggerDownload(jsonStr, `aura_luxe_${type}.json`);
  } else {
    const csvRows = [Object.keys(dataset[0]).join(',')];
    dataset.forEach(row => csvRows.push(Object.values(row).map(v => `"${v}"`).join(',')));
    const csvStr = "data:text/csv;charset=utf-8," + encodeURIComponent(csvRows.join('\n'));
    triggerDownload(csvStr, `aura_luxe_${type}.csv`);
  }
};
